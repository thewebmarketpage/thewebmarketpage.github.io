document.addEventListener('DOMContentLoaded', function() { // Load software data fetch('data.json') .then(response => response.json()) .then(data => { displayFeaturedSoftware(data.software); displayAllSoftware(data.software); setupSearch(data.software); }) .catch(error => console.error('Error loading data:', error)); // Display featured software function displayFeaturedSoftware(softwareList) { const featuredContainer = document.getElementById('featuredSoftware'); const featuredSoftware = softwareList.filter(software => software.isFeatured); featuredContainer.innerHTML = featuredSoftware.map(software => `

${software.title}

${software.description}

v${software.version} ${software.fileSize}
`).join(''); } // Display all software function displayAllSoftware(softwareList) { const allSoftwareContainer = document.getElementById('allSoftware'); allSoftwareContainer.innerHTML = softwareList.map(software => `

${software.title}

${software.description}

${software.downloads.toLocaleString()} ${software.rating}
`).join(''); } // Setup search functionality function setupSearch(softwareList) { const searchInput = document.getElementById('searchInput'); const searchButton = document.getElementById('searchButton'); const allSoftwareContainer = document.getElementById('allSoftware'); function performSearch() { const searchTerm = searchInput.value.toLowerCase(); if (searchTerm.trim() === '') { displayAllSoftware(softwareList); return; } const filteredSoftware = softwareList.filter(software => software.title.toLowerCase().includes(searchTerm) || software.description.toLowerCase().includes(searchTerm) ); allSoftwareContainer.innerHTML = filteredSoftware.map(software => `

${software.title}

${software.description}

${software.downloads.toLocaleString()} ${software.rating}
`).join(''); if (filteredSoftware.length === 0) { allSoftwareContainer.innerHTML = `

No results found for "${searchTerm}"

Try different search terms

`; } } searchButton.addEventListener('click', performSearch); searchInput.addEventListener('keyup', function(event) { if (event.key === 'Enter') { performSearch(); } }); } // Add download button functionality document.addEventListener('click', function(event) { if (event.target.classList.contains('download-btn')) { const softwareCard = event.target.closest('.software-card'); const softwareTitle = softwareCard.querySelector('h3').textContent; alert(`Downloading ${softwareTitle}...`); // In a real app, you would redirect to the download link } }); });
